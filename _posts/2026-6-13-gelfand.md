<script>
  const prefix = "{{ site.baseurl }}/img/gelfand/9b2df17d-4034-4459-a582-40a55b7c7d2e-";
  const totalPages = 49;
  const container = document.getElementById('pdf-gallery');

  // 关键：让父容器也消除行高影响（内联样式）
  container.style.lineHeight = "0";

  for (let i = 0; i < totalPages; i++) {
    const pageNum = i.toString().padStart(4, '0');
    const imgSrc = prefix + pageNum + '.webp';
    
    const picture = document.createElement('picture');
    const img = document.createElement('img');
    img.src = imgSrc;
    img.alt = `Page ${i+1}`;
    img.loading = 'lazy';
    img.classList.add('pdf-page-img');
    
    // 消除间隙的关键：图片和外层 picture 都变成块级元素
    img.style.display = "block";
    picture.style.display = "block";
    
    picture.appendChild(img);
    container.appendChild(picture);
  }
</script>
